CREATE DATABASE QuanLyKhoNhaHang;
GO


USE QuanLyKhoNhaHang;
GO

CREATE TABLE ViTri (
    MaViTri INT IDENTITY(1,1) PRIMARY KEY,
    TenViTri NVARCHAR(100) NOT NULL,
    HanMucPheDuyet DECIMAL(18,2) DEFAULT 0,
    TrangThai BIT NOT NULL DEFAULT 1
);

CREATE TABLE NguoiDung (
    MaNguoiDung INT IDENTITY(1,1) PRIMARY KEY,
    TenDangNhap VARCHAR(50) NOT NULL UNIQUE,
    MatKhau VARCHAR(255) NOT NULL,
    HoTen NVARCHAR(100) NOT NULL,
    MaViTri INT NOT NULL,
    TrangThai BIT NOT NULL DEFAULT 1,
    FOREIGN KEY (MaViTri) REFERENCES ViTri(MaViTri)
);

CREATE TABLE Kho (
    MaKho INT IDENTITY(1,1) PRIMARY KEY,
    TenKho NVARCHAR(100) NOT NULL,
    ViTriKho NVARCHAR(200)
);

CREATE TABLE DanhMucHangHoa (
    MaDanhMuc INT IDENTITY(1,1) PRIMARY KEY,
    TenDanhMuc NVARCHAR(100) NOT NULL
);

CREATE TABLE HangHoa (
    MaHangHoa INT IDENTITY(1,1) PRIMARY KEY,
    MaSKU VARCHAR(50) NOT NULL UNIQUE,
    TenHangHoa NVARCHAR(200) NOT NULL,
    MaDanhMuc INT NOT NULL,
    DonViTinh NVARCHAR(50) NOT NULL,
    TonToiThieu DECIMAL(18,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (MaDanhMuc) REFERENCES DanhMucHangHoa(MaDanhMuc)
);

CREATE TABLE NhaCungCap (
    MaNhaCungCap INT IDENTITY(1,1) PRIMARY KEY,
    TenNhaCungCap NVARCHAR(200) NOT NULL,
    SoDienThoai VARCHAR(20) NOT NULL,
    MaSoThue VARCHAR(50) NOT NULL,
    DiaChi NVARCHAR(255) NOT NULL -- Bổ sung theo yêu cầu
);

CREATE TABLE NhatKyHeThong (
    MaNhatKy INT IDENTITY(1,1) PRIMARY KEY,
    MaNguoiDung INT NOT NULL,
    HanhDong NVARCHAR(255) NOT NULL,
    BangTacDong VARCHAR(100) NOT NULL,
    ThoiGian DATETIME NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (MaNguoiDung) REFERENCES NguoiDung(MaNguoiDung)
);

CREATE TABLE KiemKeKho (
    MaKiemKe INT IDENTITY(1,1) PRIMARY KEY,
    MaKho INT NOT NULL,
    MaNguoiKiem INT NOT NULL,
    TrangThai NVARCHAR(50) NOT NULL,
    FOREIGN KEY (MaKho) REFERENCES Kho(MaKho),
    FOREIGN KEY (MaNguoiKiem) REFERENCES NguoiDung(MaNguoiDung)
);

CREATE TABLE DonMuaHang (
    MaDonMuaHang INT IDENTITY(1,1) PRIMARY KEY,
    MaSoPO VARCHAR(50) NOT NULL UNIQUE,
    MaNhaCungCap INT NOT NULL,
    NguoiTao INT NOT NULL,
    NguoiDuyet INT NULL,
    TrangThai NVARCHAR(50) NOT NULL,
    TongTien DECIMAL(18,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (MaNhaCungCap) REFERENCES NhaCungCap(MaNhaCungCap),
    FOREIGN KEY (NguoiTao) REFERENCES NguoiDung(MaNguoiDung),
    FOREIGN KEY (NguoiDuyet) REFERENCES NguoiDung(MaNguoiDung)
);

CREATE TABLE GiaoDichKho (
    MaGiaoDich INT IDENTITY(1,1) PRIMARY KEY,
    MaSoPhieu VARCHAR(50) NOT NULL UNIQUE,
    LoaiGiaoDich NVARCHAR(50) NOT NULL,
    MaKho INT NOT NULL,
    MaKhoDich INT NULL,
    MaDonMuaHang INT NULL,
    NguoiTao INT NOT NULL,
    TrangThai NVARCHAR(50) NOT NULL,
    TongGiaTri DECIMAL(18,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (MaKho) REFERENCES Kho(MaKho),
    FOREIGN KEY (MaKhoDich) REFERENCES Kho(MaKho),
    FOREIGN KEY (MaDonMuaHang) REFERENCES DonMuaHang(MaDonMuaHang),
    FOREIGN KEY (NguoiTao) REFERENCES NguoiDung(MaNguoiDung)
);

CREATE TABLE YeuCauDuyet (
    MaYeuCau INT IDENTITY(1,1) PRIMARY KEY,
    MaGiaoDich INT NOT NULL,
    NguoiYeuCau INT NOT NULL,
    NguoiDuyet INT NULL,
    GiaTriYeuCau DECIMAL(18,2) NOT NULL,
    TrangThai NVARCHAR(50) NOT NULL,
    FOREIGN KEY (MaGiaoDich) REFERENCES GiaoDichKho(MaGiaoDich),
    FOREIGN KEY (NguoiYeuCau) REFERENCES NguoiDung(MaNguoiDung),
    FOREIGN KEY (NguoiDuyet) REFERENCES NguoiDung(MaNguoiDung)
);

CREATE TABLE LoHang (
    MaLo INT IDENTITY(1,1) PRIMARY KEY,
    MaHangHoa INT NOT NULL,
    MaKho INT NOT NULL,
    SoLo VARCHAR(50) NOT NULL,
    NgaySanXuat DATE,
    HanSuDung DATE NOT NULL,
    SoLuongTon DECIMAL(18,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (MaHangHoa) REFERENCES HangHoa(MaHangHoa),
    FOREIGN KEY (MaKho) REFERENCES Kho(MaKho)
);

CREATE TABLE TonKhoTucThoi (
    MaHangHoa INT NOT NULL,
    MaKho INT NOT NULL,
    SoLuongTon DECIMAL(18,2) NOT NULL DEFAULT 0,
    NgayCapNhat DATETIME NOT NULL DEFAULT GETDATE(),
    PRIMARY KEY (MaHangHoa, MaKho),
    FOREIGN KEY (MaHangHoa) REFERENCES HangHoa(MaHangHoa),
    FOREIGN KEY (MaKho) REFERENCES Kho(MaKho)
);

CREATE TABLE CTKiemKe (
    MaCTKiemKe INT IDENTITY(1,1) PRIMARY KEY,
    MaKiemKe INT NOT NULL,
    MaHangHoa INT NOT NULL,
    MaLo INT NULL,
    SoLuongHeThong DECIMAL(18,2) NOT NULL,
    SoLuongThucTe DECIMAL(18,2) NOT NULL,
    ChenhLech DECIMAL(18,2) NOT NULL,
    FOREIGN KEY (MaKiemKe) REFERENCES KiemKeKho(MaKiemKe),
    FOREIGN KEY (MaHangHoa) REFERENCES HangHoa(MaHangHoa),
    FOREIGN KEY (MaLo) REFERENCES LoHang(MaLo)
);

CREATE TABLE CTGiaoDichKho (
    MaChiTiet INT IDENTITY(1,1) PRIMARY KEY,
    MaGiaoDich INT NOT NULL,
    MaHangHoa INT NOT NULL,
    MaLo INT NULL,
    SoLuong DECIMAL(18,2) NOT NULL,
    DonGia DECIMAL(18,2) NOT NULL,
    ThanhTien DECIMAL(18,2) NOT NULL,
    FOREIGN KEY (MaGiaoDich) REFERENCES GiaoDichKho(MaGiaoDich),
    FOREIGN KEY (MaHangHoa) REFERENCES HangHoa(MaHangHoa),
    FOREIGN KEY (MaLo) REFERENCES LoHang(MaLo)
);

CREATE TABLE CTDonMuaHang (
    MaChiTietPO INT IDENTITY(1,1) PRIMARY KEY,
    MaDonMuaHang INT NOT NULL,
    MaHangHoa INT NOT NULL,
    SoLuongDat DECIMAL(18,2) NOT NULL,
    DonGiaNhap DECIMAL(18,2) NOT NULL,
    ThanhTien DECIMAL(18,2) NOT NULL,
    FOREIGN KEY (MaDonMuaHang) REFERENCES DonMuaHang(MaDonMuaHang),
    FOREIGN KEY (MaHangHoa) REFERENCES HangHoa(MaHangHoa)
);
